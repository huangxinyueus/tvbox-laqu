    const domains = [
        {
            name: "zzztool.com",
            url: "https://www.zzztool.com/",
            description: "主要域名"
        },
        {

    async function checkDomain(url, timeout = 5000) {
        const controller = new AbortController();
        const tid = setTimeout(() => controller.abort(), timeout);
        try {
            await fetch(url, {
                method: 'HEAD',
                mode: 'no-cors',
                signal: controller.signal,
                cache: 'no-store'
            });
            clearTimeout(tid);
            return true;
        } catch {
            clearTimeout(tid);
            return false;
        }
    }
    function createCardHTML(domain, index) {
        const safeName = escapeHtml(domain.name);
        const safeUrl = escapeHtml(domain.url);
        const safeDesc = escapeHtml(domain.description || '');
        return `
            <div class="domain-card" data-index="${index}">
                <div class="card-inner">
                    <div class="card-header">
                        <div>
                            <div class="domain-title">${safeName}</div>
                            <div class="domain-desc">${safeDesc}</div>
                        </div>
                        <div class="status-badge badge-checking">
                            <span class="status-dot status-checking"></span>
                            <span class="status-text">检测中...</span>
                        </div>
                    </div>
                    <a href="${safeUrl}" target="_blank" rel="noopener noreferrer" class="visit-btn">
                        访问网站
                    </a>
                </div>
            </div>
        `;
    }
    function updateCardStatus(index, isOnline) {
        const card = document.querySelector(`.domain-card[data-index="${index}"]`);
        if (!card) return;
        const badge = card.querySelector('.status-badge');
        const dot = badge.querySelector('.status-dot');
        const textSpan = badge.querySelector('.status-text');
        if (!badge || !dot || !textSpan) return;

        dot.classList.remove('status-checking', 'status-online', 'status-offline');
        badge.classList.remove('badge-checking', 'badge-online', 'badge-offline');

        if (isOnline === true) {
            dot.classList.add('status-online');
            textSpan.innerText = '在线';
            badge.classList.add('badge-online');
        } else if (isOnline === false) {
            dot.classList.add('status-offline');
            textSpan.innerText = '离线';
            badge.classList.add('badge-offline');
        } else {
            dot.classList.add('status-checking');
            textSpan.innerText = '检测中...';
            badge.classList.add('badge-checking');
        }
    }
    function escapeHtml(str) {
        if (!str) return '';
        return str.replace(/[&<>]/g, function(m) {
            if (m === '&') return '&amp;';
            if (m === '<') return '&lt;';
            if (m === '>') return '&gt;';
            return m;
        });
    }
    async function init() {
        const container = document.getElementById('domain-list');
        if (!container) return;
        container.innerHTML = '';
        domains.forEach((domain, idx) => {
            container.insertAdjacentHTML('beforeend', createCardHTML(domain, idx));
        });
        const promises = domains.map(async (domain, idx) => {
            const online = await checkDomain(domain.url);
            updateCardStatus(idx, online);
        });
        await Promise.all(promises);
    }
    window.addEventListener('DOMContentLoaded', init);
</script>
